import { useEffect, useMemo, useRef, useState } from "react";
import "../styles/MembersSelect.css";

/**
 * MembersSelect
 * Searchable multi-select for picking users.
 *
 * Props:
 *  - options:  [{ _id, name, role?, email? }]
 *  - value:    array of selected _ids
 *  - onChange: (nextIds: string[]) => void
 *  - placeholder
 *  - maxSelected (optional) — soft limit
 */
const MembersSelect = ({
  options = [],
  value = [],
  onChange,
  placeholder = "Search and select members...",
  maxSelected,
}) => {
  const [open, setOpen] = useState(false);
  const [query, setQuery] = useState("");
  const wrapperRef = useRef(null);
  const searchRef = useRef(null);

  useEffect(() => {
    if (!open) return;

    const onPointerDown = (e) => {
      if (!wrapperRef.current?.contains(e.target)) setOpen(false);
    };
    const onKeyDown = (e) => {
      if (e.key === "Escape") setOpen(false);
    };

    document.addEventListener("mousedown", onPointerDown);
    document.addEventListener("touchstart", onPointerDown);
    document.addEventListener("keydown", onKeyDown);

    return () => {
      document.removeEventListener("mousedown", onPointerDown);
      document.removeEventListener("touchstart", onPointerDown);
      document.removeEventListener("keydown", onKeyDown);
    };
  }, [open]);

  useEffect(() => {
    if (open) {
      setQuery("");
      const t = setTimeout(() => searchRef.current?.focus(), 0);
      return () => clearTimeout(t);
    }
  }, [open]);

  const filtered = useMemo(() => {
    const q = query.trim().toLowerCase();
    if (!q) return options;
    return options.filter(
      (u) =>
        u.name?.toLowerCase().includes(q) ||
        u.email?.toLowerCase().includes(q) ||
        u.role?.toLowerCase().includes(q)
    );
  }, [options, query]);

  const selectedSet = useMemo(() => new Set(value), [value]);
  const selectedOptions = useMemo(
    () => options.filter((u) => selectedSet.has(u._id)),
    [options, selectedSet]
  );

  const toggle = (id) => {
    if (selectedSet.has(id)) {
      onChange?.(value.filter((v) => v !== id));
    } else {
      if (maxSelected && value.length >= maxSelected) return;
      onChange?.([...value, id]);
    }
  };

  const removeOne = (id) => {
    onChange?.(value.filter((v) => v !== id));
  };

  return (
    <div className="members-select" ref={wrapperRef}>
      {/* Trigger */}
      <button
        type="button"
        className={`members-select-trigger ${open ? "is-open" : ""}`}
        aria-haspopup="listbox"
        aria-expanded={open}
        onClick={() => setOpen((v) => !v)}
      >
        <span className="members-select-value">
          {selectedOptions.length === 0 ? (
            <span className="members-select-placeholder">{placeholder}</span>
          ) : (
            <span className="members-select-count">
              {selectedOptions.length} selected
            </span>
          )}
        </span>
        <span className="members-select-caret" aria-hidden="true">▾</span>
      </button>

      {/* Selected chips */}
      {selectedOptions.length > 0 && (
        <div className="members-select-chips">
          {selectedOptions.map((u) => (
            <span className="members-chip" key={u._id}>
              <span className="members-chip-name">{u.name}</span>
              <button
                type="button"
                className="members-chip-remove"
                aria-label={`Remove ${u.name}`}
                onClick={() => removeOne(u._id)}
              >
                ×
              </button>
            </span>
          ))}
        </div>
      )}

      {/* Dropdown panel */}
      {open && (
        <div className="members-select-panel" role="listbox">
          <div className="members-select-search">
            <input
              ref={searchRef}
              type="search"
              value={query}
              onChange={(e) => setQuery(e.target.value)}
              placeholder="Search by name, email or role..."
              aria-label="Search members"
            />
            {query && (
              <button
                type="button"
                className="members-select-clear"
                aria-label="Clear search"
                onClick={() => setQuery("")}
              >
                ×
              </button>
            )}
          </div>

          <div className="members-select-list">
            {filtered.length === 0 ? (
              <p className="members-select-empty">No members match "{query}".</p>
            ) : (
              filtered.map((u) => {
                const checked = selectedSet.has(u._id);
                return (
                  <label
                    key={u._id}
                    className={`members-select-option ${checked ? "is-checked" : ""}`}
                  >
                    <input
                      type="checkbox"
                      checked={checked}
                      onChange={() => toggle(u._id)}
                    />
                    <span className="members-select-avatar">
                      {u.name?.charAt(0).toUpperCase()}
                    </span>
                    <span className="members-select-info">
                      <strong>{u.name}</strong>
                      {u.email && <small>{u.email}</small>}
                    </span>
                    {u.role && (
                      <span className={`members-select-role role-${u.role}`}>
                        {u.role}
                      </span>
                    )}
                  </label>
                );
              })
            )}
          </div>

          <div className="members-select-footer">
            <span>{value.length} selected</span>
            {value.length > 0 && (
              <button
                type="button"

                onClick={() => onChange?.([])}

                Clear all

import { createContext, useContext, useEffect, useState } from "react";
import API from "../data/api";

const AuthContext = createContext(null);

export const AuthProvider = ({ children }) => {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  const checkAuth = async () => {
    try {
      const response = await fetch(
        `${API}/api/auth/me`,
        {
          method: "GET",
          credentials: "include",
        }
      );

      if (response.status === 401) {
        setUser(null);
        return null;
      }

      if (!response.ok) {
        throw new Error("Failed to verify authentication.");
      }

      const data = await response.json();

      setUser(data.user);
      return data.user;
    } catch (error) {
      console.error("Auth check failed:", error.message);
      setUser(null);
      return null;
    } finally {
      setLoading(false);
    }
  };

  const handleLogout = async () => {
    try {
      const response = await fetch(
        `${API}/api/auth/logout`,
        {
          method: "POST",
          credentials: "include",
        }
      );

      const data = await response.json();

      if (!response.ok) {
        throw new Error(data.message || "Logout failed.");
      }

      setUser(null);
      navigate("/login", { replace: true });
    } catch (error) {
      console.error("Logout error:", error.message);
    }
  }

  useEffect(() => {
    checkAuth();
  }, []);

  return (
    <AuthContext.Provider value={{ user, setUser, loading, checkAuth, handleLogout }}>
      {children}
    </AuthContext.Provider>
  );
};

export const useAuth = () => {
  const context = useContext(AuthContext);

  if (!context) {
    throw new Error("useAuth must be used inside AuthProvider");
  }

  return context;
};

    if (!formData.user) {
      setError("Please select a user.");
      return;
    }

    setLoading(true);

    try {
      const payload = {
        user: formData.user,
        projects: formData.projects,
      };

      const response = editingId
        ? await axios.patch(
          `${API}/api/permissions/${editingId}`,
          { projects: payload.projects },
          { withCredentials: true }
        )
        : await axios.post(
          `${API}/api/permissions`,
          payload,
          { withCredentials: true }
        );

      setSuccess(
        response.data.message ||
        (editingId
          ? "Permission updated successfully."
          : "Permission created successfully.")
      );

      resetForm();
      revalidator.revalidate();
    } catch (err) {
      setError(
        err.response?.data?.message ||
        "Failed to save permission."
      );
    } finally {
      setLoading(false);
    }
  };

  const handleEdit = (permission) => {
    const userId =
      permission.user?._id || permission.user?.id || permission.user;

    const projectIds = (permission.projects ?? []).map(
      (project) =>
        typeof project === "object"
          ? project._id || project.id
          : project
    );

    setEditingId(permission._id || permission.id);

    setFormData({
      user: userId || "",
      projects: projectIds,
    });

    setError("");
    setSuccess("");

    window.scrollTo({ top: 0, behavior: "smooth" });
  };

  const handleDelete = async (id) => {
    if (
      !window.confirm(
        "Are you sure you want to delete this permission?"
      )
    ) {
      return;
    }

    setError("");
    setSuccess("");

    try {
      const response = await axios.delete(
        `${API}/api/permissions/${id}`,
        { withCredentials: true }
      );

      setSuccess(
        response.data.message ||
        "Permission deleted successfully."
      );

      if (editingId === id) {
        resetForm();
      }

      revalidator.revalidate();
    } catch (err) {
      setError(
        err.response?.data?.message ||
        "Failed to delete permission."
      );
    }
  };

  const filteredPermissions = useMemo(() => {
    const query = search.toLowerCase();

    return permissions.filter((permission) => {
      const user =
        typeof permission.user === "object"
          ? permission.user
          : users.find(
            (item) =>
              (item._id || item.id) === permission.user
          );

      const userName = user?.name ?? "";
      const email = user?.email ?? "";

      const projectNames = (permission.projects ?? [])
        .map((project) => {
          if (typeof project === "object") {
            return project.title || project.name || "";
          }

          const found = projects.find(
            (item) =>
              (item._id || item.id) === project
          );

          return found?.title || found?.name || "";
        })
        .join(" ");

      return (
        userName.toLowerCase().includes(query) ||
        email.toLowerCase().includes(query) ||
        projectNames.toLowerCase().includes(query)
      );
    });
  }, [permissions, users, projects, search]);

  return (
    <main className="timesheet-container">
      <section className="timesheet-form-section">
        <div className="section-header">
          <h1>
            {editingId
              ? "Update Permission"
              : "Create Permission"}
          </h1>
          <p>
            Assign users access to one or more projects.
          </p>
        </div>

        <form
          onSubmit={handleSubmit}
          className="timesheet-form"
        >
          <div className="form-group">
            <label htmlFor="permission-user">User</label>

            <SearchableSelect
              options={users}
              value={formData.user}
              onChange={handleUserChange}
              labelKey="name"
              valueKey="_id"
              secondaryKey="email"
              placeholder="Select User"
              searchPlaceholder="Search users..."
            />
          </div>



          <div className="form-group permission-projects">
            <div className="permission-project-header">
              <div>
                <label className="permission-project-title">
                  Assign Projects
                </label>
                <p className="permission-project-subtitle">
                  Search and select projects this user can access.
                </p>
              </div>

              <span className="permission-selected-count">
                {formData.projects.length} selected
              </span>
            </div>

            <ProjectsSelect
              options={projects}
              value={formData.projects}
              onChange={(ids) =>
                setFormData((prev) => ({
                  ...prev,
                  projects: ids,
                }))
              }
            />

            <div className="permission-project-footer">
              <span>
                {formData.projects.length === 0
                  ? "No projects selected"
                  : `${formData.projects.length} project(s) assigned`}
              </span>
            </div>
          </div>



          {error && (
            <p className="timesheet-error" role="alert">
              {error}
            </p>
          )}

          {success && (
            <p className="timesheet-success" role="status">
              {success}
            </p>
          )}

          <div className="timesheet-actions">
            <button
              type="submit"
              className="submit-btn"
              disabled={loading}
            >
              {loading
                ? "Saving..."
                : editingId
                  ? "Update Permission"
                  : "Create Permission"}
            </button>

            {editingId && (
              <button
                type="button"
                className="cancel-btn"
                onClick={resetForm}
              >
                Cancel Edit
              </button>
            )}
          </div>
        </form>
      </section>

      <section className="task-list-section">
        <div className="section-header">
          <h2>Permission List</h2>
          <p>Manage user project access.</p>
        </div>

        <div className="task-toolbar">
          <input
            type="search"
            placeholder="Search users or projects..."
            aria-label="Search permissions"
            value={search}
            onChange={(e) => setSearch(e.target.value)}
          />
        </div>

        <div className="task-table-wrapper">
          <table className="task-table">
            <thead>
              <tr>
                <th>User</th>
                <th>Email</th>
                <th>Assigned Projects</th>
                <th>Total Projects</th>
                <th>Actions</th>
              </tr>
            </thead>

            <tbody>
              {filteredPermissions.length > 0 ? (
                filteredPermissions.map((permission) => {
                  const id =
                    permission._id || permission.id;

                  const user =
                    typeof permission.user === "object"
                      ? permission.user
                      : users.find(
                        (item) =>
                          (item._id || item.id) ===
                          permission.user
                      );

                  const projectNames = (
                    permission.projects ?? []
                  ).map((project) => {
                    if (typeof project === "object") {
                      return (
                        project.title ||
                        project.name ||
                        "Untitled Project"
                      );
                    }

                    const found = projects.find(
                      (item) =>
                        (item._id || item.id) === project
                    );

                    return (
                      found?.title ||
                      found?.name ||
                      "Unknown Project"
                    );
                  });

                  return (
                    <tr key={id}>
                      <td>{user?.name || "Unknown User"}</td>
                      <td>{user?.email || "—"}</td>
                      <td>
                        {projectNames.length
                          ? projectNames.join(", ")
                          : "No projects assigned"}
                      </td>
                      <td>{projectNames.length}</td>
                      <td>
                        <div className="task-row-actions">
                          <button
                            type="button"
                            onClick={() =>
                              handleEdit(permission)
                            }
                          >
                            Edit
                          </button>

                          <button
                            type="button"
                            onClick={() => handleDelete(id)}
                          >
                            Delete
                          </button>
                        </div>
                      </td>
                    </tr>
                  );
                })
              ) : (
                <tr>
                  <td
                    colSpan="5"
                    className="empty-task-list"
                  >
                    {permissions.length === 0
                      ? "No permissions created yet."
                      : "No matching permissions found."}
                  </td>
                </tr>
              )}
            </tbody>
          </table>
        </div>
      </section>
    </main>
  );
};

export default Permission;

import { useEffect, useMemo, useState } from "react";
import {
  useLoaderData,
  useRevalidator,
  useSearchParams,
} from "react-router-dom";
import axios from "axios";
import "../styles/Project.css";
import MembersSelect from "../components/MembersSelect";
import API from "../data/api";

const initialFormData = {
  title: "",
  description: "",
  type: "non-billable",
  status: "planned",
  clientName: "",
  manager: "",
  members: [],
  startDate: "",
  dueDate: "",
  totalHours: "",
};

const Project = () => {
  const data = useLoaderData();
  const revalidator = useRevalidator();
  const [searchParams, setSearchParams] = useSearchParams();

  const users = Array.isArray(data?.users) ? data.users : [];
  const pagination = data.pagination || {};

  const currentPage = Math.max(1, Number(searchParams.get("page")) || pagination.currentPage || 1);
  const limit = Math.max(1, Number(searchParams.get("limit")) || pagination.limit || 10);

  const totalProjects =
    pagination.totalProjects ??
    pagination.total ??
    (Array.isArray(data?.projects) ? data.projects.length : 0);

  const totalPage = Math.max(1, Math.ceil(totalProjects / limit));

  const getUserName = (userId) => {
    const id = typeof userId === "object" ? userId?._id : userId;
    return users.find((user) => user._id === id)?.name || "Unknown user";
  };

  const decorateProjects = (list) =>
    list.map((project) => ({
      ...project,
      managerName: getUserName(project.manager),
      memberNames: (project.members ?? []).map(getUserName),
    }));

  const [projectList, setProjectList] = useState(() =>
    decorateProjects(Array.isArray(data?.projects) ? data.projects : [])
  );

  const [formData, setFormData] = useState(initialFormData);
  const [editingId, setEditingId] = useState(null);
  const [error, setError] = useState("");
  const [success, setSuccess] = useState("");
  const [loading, setLoading] = useState(false);
  const [search, setSearch] = useState("");
  const [statusFilter, setStatusFilter] = useState("all");

  useEffect(() => {
    setProjectList(decorateProjects(Array.isArray(data?.projects) ? data.projects : []));
  }, [data]);

  const managers = users.filter((user) => ["manager", "admin"].includes(user.role));
  const employees = users.filter((user) => user.role === "employee");

  const handleChange = (e) => {
    const { name, value } = e.target;
    setFormData((prev) => ({ ...prev, [name]: value }));
    setError("");
    setSuccess("");
  };

  const resetForm = () => {
    setFormData(initialFormData);
    setEditingId(null);
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError("");
    setSuccess("");

    if (formData.dueDate && formData.dueDate < formData.startDate) {
      setError("Due date cannot be before the start date.");
      return;
    }

    setLoading(true);

    try {
      const payload = {
        ...formData,
        totalHours: Number(formData.totalHours),
      };

      const response = editingId
        ? await axios.put(`${API}/api/project/${editingId}`, payload, {
          withCredentials: true,
        })
        : await axios.post(`${API}/api/project`, payload, {
          withCredentials: true,
        });

      setSuccess(
        response.data.message ||
        (editingId ? "Project updated successfully." : "Project created successfully.")
      );

      resetForm();
      revalidator.revalidate();
    } catch (err) {
      setError(err.response?.data?.message || "Failed to save project.");
    } finally {
      setLoading(false);
    }
  };

  const handleEdit = (project) => {
    const projectId = project._id || project.id;
    setEditingId(projectId);

    setFormData({
      title: project.title ?? "",
      description: project.description ?? "",
      type: project.type ?? "non-billable",
      status: project.status ?? "planned",
      clientName: project.clientName ?? "",
      manager: project.manager?._id ?? project.manager?.id ?? project.manager ?? "",
      members: (project.members ?? []).map((m) =>
        typeof m === "object" ? m._id ?? m.id : m
      ),
      startDate: project.startDate ? String(project.startDate).slice(0, 10) : "",
      dueDate: project.dueDate ? String(project.dueDate).slice(0, 10) : "",
      totalHours: project.totalHours != null ? String(project.totalHours) : "",
    });

    setError("");
    setSuccess("");

    window.scrollTo({ top: 0, behavior: "smooth" });
  };

  const handleDelete = async (id) => {
    const confirmed = window.confirm(
      "Are you sure you want to delete this project?"
    );

    if (!confirmed) return;

    setError("");
    setSuccess("");

    try {
      const response = await axios.delete(`${API}/api/project/${id}`, {
        withCredentials: true,
      });

      setSuccess(response.data.message || "Project deleted successfully.");

      if (editingId === id) {
        resetForm();
      }

      revalidator.revalidate();
    } catch (error) {
      setError(error.response?.data?.message || "Failed to delete project.");
      console.error("Delete Project Error:", error.response?.data || error.message);
    }
  };

  const handlePageChange = (page) => {
    if (page < 1 || page > totalPage || page === currentPage) return;

    setSearchParams((prev) => {
      const params = new URLSearchParams(prev);
      params.set("page", String(page));
      params.set("limit", String(limit));
      return params;
    });
  };

  const handleLimitChange = (e) => {
    const newLimit = Number(e.target.value);

    setSearchParams((prev) => {
      const params = new URLSearchParams(prev);
      params.set("page", "1");
      params.set("limit", String(newLimit));
      return params;
    });
  };

  const filteredProjects = useMemo(() => {
    const query = search.toLowerCase();

    return projectList.filter((project) => {
      const matchesSearch =
        (project.title ?? "").toLowerCase().includes(query) ||
        (project.clientName ?? "").toLowerCase().includes(query);

      const matchesStatus = statusFilter === "all" || project.status === statusFilter;

      return matchesSearch && matchesStatus;
    });
  }, [projectList, search, statusFilter]);

  const pageHours = projectList.reduce((total, project) => total + Number(project.totalHours || 0), 0);

  return (
    <main className="project-container">
      <section className="project-form-section">
        <div className="project-section-header">
          <h1>{editingId ? "Update Project" : "Create Project"}</h1>
          <p>
            {editingId
              ? "Edit the project details and save your changes."
              : "Add a new project and assign team members."}
          </p>
        </div>

        <form className="project-form" onSubmit={handleSubmit}>
          <div className="project-form-group">
            <label htmlFor="title">Project Title</label>
            <input
              id="title"
              name="title"
              type="text"
              placeholder="Enter project title"
              value={formData.title}
              onChange={handleChange}
              required
            />
          </div>

          <div className="project-form-group">
            <label htmlFor="description">Description</label>
            <textarea
              id="description"
              name="description"
              rows="3"
              placeholder="Describe the project..."
              value={formData.description}
              onChange={handleChange}
            />
          </div>

          <div className="project-form-row">
            <div className="project-form-group">
              <label htmlFor="type">Project Type</label>
              <select
                id="type"
                name="type"
                value={formData.type}
                onChange={handleChange}
                required
              >
                <option value="billable">Billable</option>
                <option value="non-billable">Non-billable</option>
                <option value="internal">Internal</option>
              </select>
            </div>

            <div className="project-form-group">
              <label htmlFor="status">Status</label>
              <select
                id="status"
                name="status"
                value={formData.status}
                onChange={handleChange}
                required
              >
                <option value="planned">Planned</option>
                <option value="active">Active</option>
                <option value="on-hold">On Hold</option>
                <option value="completed">Completed</option>
                <option value="cancelled">Cancelled</option>
              </select>
            </div>
          </div>

          <div className="project-form-group">
            <label htmlFor="clientName">Client Name</label>
            <input
              id="clientName"
              name="clientName"
              type="text"
              placeholder="Enter client name"
              value={formData.clientName}
              onChange={handleChange}
            />
          </div>

          <div className="project-form-group">
            <label htmlFor="manager">Project Manager</label>
            <select
              id="manager"
              name="manager"
              value={formData.manager}
              onChange={handleChange}
              required
            >
              <option value="">Select Manager</option>
              {managers.map((user) => (
                <option key={user._id} value={user._id}>
                  {user.name} ({user.role})
                </option>
              ))}
            </select>

            {managers.length === 0 && (
              <small>No managers or admins available.</small>
            )}
          </div>

          <div className="project-form-group">
            <label htmlFor="members">Assign Members</label>
            <MembersSelect
              options={employees}
              value={formData.members}
              onChange={(ids) => {
                setFormData((prev) => ({ ...prev, members: ids }));
                setError("");
                setSuccess("");
              }}
              placeholder="Search and select members..."
            />
            <small>
              Type to filter. Click a member to add or remove them.
            </small>
          </div>

          <div className="project-form-row">
            <div className="project-form-group">
              <label htmlFor="startDate">Start Date</label>
              <input
                id="startDate"
                name="startDate"
                type="date"
                value={formData.startDate}
                onChange={handleChange}
                required
              />
            </div>

            <div className="project-form-group">
              <label htmlFor="dueDate">Due Date</label>
              <input
                id="dueDate"
                name="dueDate"
                type="date"
                min={formData.startDate || undefined}
                value={formData.dueDate}
                onChange={handleChange}
              />
            </div>
          </div>

          <div className="project-form-group">
            <label htmlFor="totalHours">Total Allocated Hours</label>
            <input
              id="totalHours"
              name="totalHours"
              type="number"
              min="1"
              step="0.5"
              placeholder="e.g. 500"
              value={formData.totalHours}
              onChange={handleChange}
              required
            />
          </div>

          {error && (
            <p className="project-error" role="alert">
              {error}
            </p>
          )}

          {success && (
            <p className="project-success" role="status">
              {success}
            </p>
          )}

          <div className="project-actions">
            <button type="submit" className="project-submit-btn" disabled={loading}>
              {loading
                ? "Saving..."
                : editingId
                  ? "Update Project"
                  : "Create Project"}
            </button>

            {editingId && (
              <button
                type="button"
                className="project-cancel-btn"
                onClick={resetForm}
                disabled={loading}
              >
                Cancel Edit
              </button>
            )}
          </div>
        </form>
      </section>

      <section className="project-list-section">
        <div className="project-section-header project-list-header">
          <div>
            <h2>Project List</h2>
            <p>Manage your projects and assigned team members.</p>
          </div>
        </div>

        <div className="project-toolbar">
          <input
            type="search"
            placeholder="Search by title or client..."
            aria-label="Search projects"
            value={search}
            onChange={(e) => setSearch(e.target.value)}
          />

          <select
            aria-label="Filter projects by status"
            value={statusFilter}
            onChange={(e) => setStatusFilter(e.target.value)}
          >
            <option value="all">All Statuses</option>
            <option value="planned">Planned</option>
            <option value="active">Active</option>
            <option value="on-hold">On Hold</option>
            <option value="completed">Completed</option>
            <option value="cancelled">Cancelled</option>
          </select>
        </div>

        <div className="project-table-wrapper">
          <table className="project-table">
            <thead>
              <tr>
                <th>Project</th>
                <th>Type</th>
                <th>Status</th>
                <th>Manager</th>
                <th>Timeline</th>
                <th>Hours</th>
                <th>Action</th>
              </tr>
            </thead>

            <tbody>
              {filteredProjects.length > 0 ? (
                filteredProjects.map((project) => {
                  const projectId = project._id || project.id;
                  const isEditing = editingId === projectId;

                  return (
                    <tr
                      key={projectId}
                      className={isEditing ? "is-editing" : undefined}
                    >
                      <td>
                        <div className="project-table-title">{project.title}</div>
                        <small>{project.clientName || "No client"}</small>
                      </td>

                      <td>
                        <span className="project-type-badge">{project.type}</span>
                      </td>

                      <td>
                        <span
                          className={`project-status-badge status-${project.status}`}
                        >
                          {project.status?.replace("-", " ") || "Unknown"}
                        </span>
                      </td>

                      <td>{project.managerName || getUserName(project.manager)}</td>

                      <td>
                        <div className="project-timeline">
                          <span>{project.startDate?.slice(0, 10) || "—"}</span>
                          <span>{project.dueDate?.slice(0, 10) || "No due date"}</span>
                        </div>
                      </td>

                      <td>{project.totalHours ?? 0} hrs</td>

                      <td>
                        <details className="row-menu">
                          <summary className="row-menu-trigger" aria-label="Open menu">
                            <span></span><span></span><span></span>
                          </summary>

                          <div className="row-menu-panel" role="menu">
                            <button
                              type="button"
                              className="row-menu-item"
                              role="menuitem"
                              onClick={() => handleEdit(project)}
                            >
                              <span className="row-menu-item-icon">✏️</span>
                              <span>Edit project</span>
                            </button>

                            <button
                              type="button"
                              className="row-menu-item"
                              role="menuitem"
                              onClick={() => {/* view details */ }}
                            >
                              <span className="row-menu-item-icon">📄</span>
                              <span>View details</span>
                            </button>

                            <button
                              type="button"
                              className="row-menu-item"
                              role="menuitem"
                              onClick={() => {/* assign team */ }}
                            >
                              <span className="row-menu-item-icon">👥</span>
                              <span>Assign team</span>
                            </button>

                            <button
                              type="button"
                              className="row-menu-item is-danger"
                              role="menuitem"
                              onClick={() => handleDelete(projectId)}
                            >
                              <span className="row-menu-item-icon">🗑️</span>
                              <span>Delete</span>
                            </button>
                          </div>
                        </details>
                      </td>
                    </tr>
                  );
                })
              ) : (
                <tr>
                  <td colSpan="7" className="project-empty-state">
                    <div className="project-empty-icon">📁</div>
                    <strong>
                      {projectList.length === 0
                        ? "No projects yet"
                        : "No matching projects"}
                    </strong>
                    <p>
                      {projectList.length === 0
                        ? "Create your first project using the form."
                        : "Try changing your search or status filter."}
                    </p>
                  </td>
                </tr>
              )}
            </tbody>
          </table>
        </div>

        <div className="project-summary">
          <span>Total Projects: {totalProjects}</span>
          <span>Allocated Hours (this page): {pageHours} hrs</span>

          <div className="pagination-controls">
            <label htmlFor="projects-limit">Rows per page:</label>

            <select
              id="projects-limit"
              value={limit}
              onChange={handleLimitChange}
            >
              <option value={5}>5</option>
              <option value={10}>10</option>
              <option value={20}>20</option>
              <option value={50}>50</option>
            </select>

            <button
              type="button"
              disabled={currentPage <= 1}
              onClick={() => handlePageChange(currentPage - 1)}
            >
              Previous
            </button>

            <span>Page {currentPage} of {totalPage}</span>

            <button
              type="button"
              disabled={currentPage >= totalPage}
              onClick={() => handlePageChange(currentPage + 1)}
            >
              Next
            </button>
          </div>
        </div>
      </section>
    </main>
  );
};

import { useEffect, useMemo, useState } from "react";
import {
  useLoaderData,
  useRevalidator,
  useSearchParams,
} from "react-router-dom";
import axios from "axios";
import API from "../data/api.js";
import "../styles/TimesheetForm.css";
import SearchableSelect from "../components/SearchableSelect";

const initialFormData = { title: "", project: "", employee: "", type: "development", status: "planned", date: new Date().toISOString().slice(0, 10), hours: "", description: "" };

const taskTypes = ["development", "bug-fix", "testing", "meeting", "research", "documentation", "deployment", "other"];
const taskStatuses = ["planned", "active", "on-hold", "completed", "cancelled"];

const TimesheetForm = () => {
  const data = useLoaderData();
  const revalidator = useRevalidator();
  const [searchParams, setSearchParams] = useSearchParams();

  const users = data?.users ?? [];
  const projects = data?.projects ?? [];
  const pagination = data?.pagination ?? {};

  const currentPage = Math.max(1, Number(searchParams.get("page")) || pagination.currentPage || 1);
  const limit = Math.max(1, Number(searchParams.get("limit")) || pagination.limit || 10);

  const totalTasks = pagination.totalTasks ?? data?.tasks?.length ?? 0;
  const totalPages = Math.max(1, Math.ceil(totalTasks / limit));

  const [tasks, setTasks] = useState(data?.tasks ?? []);
  const [formData, setFormData] = useState(initialFormData);
  const [editingId, setEditingId] = useState(null);
  const [search, setSearch] = useState("");
  const [error, setError] = useState("");
  const [success, setSuccess] = useState("");
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    setTasks(data?.tasks ?? []);
  }, [data]);

  const handleChange = (e) => {
    const { name, value } = e.target;
    setFormData((prev) => ({ ...prev, [name]: value }));
    setError("");
    setSuccess("");
  };

  const resetForm = () => {
    setFormData(initialFormData);
    setEditingId(null);
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError("");
    setSuccess("");

    const hours = Number(formData.hours);

    if (
      !Number.isFinite(hours) ||
      hours < 0.25 ||
      hours > 24 ||
      !Number.isInteger(hours * 4)
    ) {
      setError("Hours must be between 0.25 and 24 in increments of 0.25.");
      return;
    }

    setLoading(true);

    try {
      const payload = { ...formData, hours };

      const response = editingId
        ? await axios.put(`${API}/api/tasks/${editingId}`, payload, {
          withCredentials: true,
        })
        : await axios.post(`${API}/api/tasks`, payload, {
          withCredentials: true,
        });

      setSuccess(
        response.data.message || (editingId ? "Task updated successfully." : "Task created successfully.")
      );

      resetForm();
      revalidator.revalidate();
    } catch (err) {
      setError(err.response?.data?.message || "Failed to save task.");
    } finally {
      setLoading(false);
    }
  };

  const handleEdit = (task) => {
    const taskId = task._id || task.id;
    setEditingId(taskId);

    setFormData({
      title: task.title ?? "",
      project: task.project?._id ?? task.project?.id ?? task.project ?? "",
      employee: task.employee?._id ?? task.employee?.id ?? task.employee ?? "",
      type: task.type ?? "development",
      status: task.status ?? "planned",
      date: task.date ? task.date.slice(0, 10) : "",
      hours: String(task.hours ?? ""),
      description: task.description ?? "",
    });

    setError("");
    setSuccess("");

    window.scrollTo({ top: 0, behavior: "smooth" });
  };

  const handleDelete = async (id) => {
    if (!window.confirm("Are you sure you want to delete this task?")) {
      return;
    }

    setError("");
    setSuccess("");

    try {
      const response = await axios.delete(`${API}/api/tasks/${id}`, {
        withCredentials: true,
      });

      setSuccess(response.data.message || "Task deleted successfully.");

      if (editingId === id) {
        resetForm();
      }

      revalidator.revalidate();
    } catch (err) {
      setError(err.response?.data?.message || "Failed to delete task.");
    }
  };

  const handlePageChange = (page) => {
    if (page < 1 || page > totalPages || page === currentPage) {
      return;
    }

    setSearchParams((prev) => {
      const params = new URLSearchParams(prev);
      params.set("page", String(page));
      params.set("limit", String(limit));
      return params;
    });
  };

  const handleLimitChange = (e) => {
    setSearchParams((prev) => {
      const params = new URLSearchParams(prev);
      params.set("page", "1");
      params.set("limit", e.target.value);
      return params;
    });
  };

  const filteredTasks = useMemo(() => {
    const query = search.toLowerCase();

    return tasks.filter((task) => {
      const taskProjectId = task.project?._id || task.project?.id || task.project;
      const taskEmployeeId = task.employee?._id || task.employee?.id || task.employee;

      const projectName =
        typeof task.project === "object" && task.project?.title
          ? task.project.title
          : projects.find((project) => (project._id || project.id) === taskProjectId)?.title;

      const employeeName =
        typeof task.employee === "object" && task.employee?.name
          ? task.employee.name
          : users.find((user) => (user._id || user.id) === taskEmployeeId)?.name;

      return (
        (task.title ?? "").toLowerCase().includes(query) ||
        (task.description ?? "").toLowerCase().includes(query) ||
        (projectName ?? "").toLowerCase().includes(query) ||
        (employeeName ?? "").toLowerCase().includes(query)
      );
    });
  }, [tasks, search, projects, users]);

  const pageHours = tasks.reduce((total, task) => total + Number(task.hours || 0), 0);

  return (
    <main className="timesheet-container">
      <section className="timesheet-form-section">
        <div className="section-header">
          <h1>{editingId ? "Update Task" : "Create Task"}</h1>
          <p>Record and manage your daily work activities.</p>
        </div>

        <form onSubmit={handleSubmit} className="timesheet-form">
          {/* <div className="form-group">
            <label htmlFor="title">Task Title</label>
            <input
              id="title"
              name="title"
              type="text"
              maxLength={160}
              placeholder="Enter task title"
              value={formData.title}
              onChange={handleChange}
            />
          </div> */}

          <div className="form-group">
            <label htmlFor="project">Project</label>
            {/* <select
              id="project"
              name="project"
              value={formData.project}
              onChange={handleChange}
              required
            >
              <option value="">Select Project</option>
              {projects.map((project) => (
                <option key={project._id} value={project._id}>
                  {project.title}
                </option>
              ))}
            </select> */}
            <SearchableSelect
              options={projects}
              value={formData.project}
              onChange={(id) => {
                setFormData((prev) => ({ ...prev, project: id }));
                setError("");
                setSuccess("");
              }}
              labelKey="title"
              valueKey="_id"
              placeholder="Select Project"
              searchPlaceholder="Search projects..."
            />
          </div>

          <div className="form-group">
            <label htmlFor="employee">Employee</label>
            {/* <select
              id="employee"
              name="employee"
              value={formData.employee}
              onChange={handleChange}
              required
            >
              <option value="">Select Employee</option>
              {users
                .filter((user) => user.role === "employee")
                .map((user) => (
                  <option key={user._id} value={user._id}>
                    {user.name} ({user.email})
                  </option>
                ))}
            </select> */}
            <SearchableSelect
              options={users.filter((user) => user.role === "employee")}
              value={formData.employee}
              onChange={(id) => {
                setFormData((prev) => ({ ...prev, employee: id }));
                setError("");
                setSuccess("");
              }}
              labelKey="name"
              valueKey="_id"
              secondaryKey="email"
              placeholder="Select Employee"
              searchPlaceholder="Search employees..."
            />
          </div>

          <div className="form-group">
            <label htmlFor="type">Task Type</label>
            <select
              id="type"
              name="type"
              value={formData.type}
              onChange={handleChange}
              required
            >
              {taskTypes.map((type) => (
                <option key={type} value={type}>
                  {type.replace("-", " ")}
                </option>
              ))}
            </select>
          </div>

          <div className="form-group">
            <label htmlFor="status">Status</label>
            <select
              id="status"
              name="status"
              value={formData.status}
              onChange={handleChange}
              required
            >
              {taskStatuses.map((status) => (
                <option key={status} value={status}>
                  {status.replace("-", " ")}
                </option>
              ))}
            </select>
          </div>

          <div className="form-group">

          <div className="form-group">
            <label htmlFor="description">Description</label>
            <textarea

              placeholder="Describe the work completed..."

import { useEffect, useMemo, useState } from "react";
import "../styles/User.css";
import { useSearchParams, useLoaderData, useRevalidator } from "react-router-dom";
import axios from "axios";
import API from "../data/api";

const initialFormData = {
  name: "",
  email: "",
  password: "",
  role: "employee",
};

const User = () => {
  const data = useLoaderData();

  const [formData, setFormData] = useState(initialFormData);
  const [users, setUsers] = useState(data.users || []);
  const pagination = data.pagination || {};
  const [search, setSearch] = useState("");
  const [roleFilter, setRoleFilter] = useState("all");
  const [error, setError] = useState("");
  const [success, setSuccess] = useState("");
  const [editingId, setEditingId] = useState(null);
  const [loading, setLoading] = useState(false);
  const [searchParams, setSearchParams] = useSearchParams();
  const currentPage = Math.max(1, Number(searchParams.get("page")) || pagination.currentPage || 1);
  const limit = Math.max(1, Number(searchParams.get("limit")) || pagination.limit || 10);
  const totalUsers = pagination.totalUsers ?? users.length;
  const totalPage = Math.max(1, Math.ceil(totalUsers / limit));
  const revalidator = useRevalidator();

  useEffect(() => {
    setUsers(data.users || []);
  }, [data]);

  const handleChange = (e) => {
    const { name, value } = e.target;
    setFormData((prev) => ({ ...prev, [name]: value }));
    setError("");
    setSuccess("");
  };

  const resetForm = () => {
    setFormData(initialFormData);
    setEditingId(null);
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError("");
    setSuccess("");

    const name = formData.name.trim();
    const email = formData.email.trim().toLowerCase();

    if (!name || !email || (!editingId && !formData.password)) {
      setError("Please fill in all required fields.");
      return;
    }

    if (formData.password && formData.password.length < 8) {
      setError("Password must contain at least 8 characters.");
      return;
    }

    const emailTaken = users.some(
      (user) =>
        user.email.toLowerCase() === email &&
        (user._id || user.id) !== editingId
    );

    if (emailTaken) {
      setError("A user with this email already exists.");
      return;
    }

    setLoading(true);

    try {
      const payload = {
        name,
        email,
        role: formData.role,
      };

      if (formData.password) {
        payload.password = formData.password;
      }

      const response = editingId
        ? await axios.put(`${API}/api/user/${editingId}`, payload, {
          withCredentials: true,
        })
        : await axios.post(`${API}/api/auth/register`, payload, {
          withCredentials: true,
        });

      setSuccess(
        response.data.message ||
        (editingId ? "User updated successfully." : "User created successfully.")
      );

      resetForm();
      revalidator.revalidate();
    } catch (err) {
      setError(
        err.response?.data?.message ||
        (editingId ? "Failed to update user." : "Failed to create user.")
      );
    } finally {
      setLoading(false);
    }
  };

  const handleEdit = (user) => {
    const userId = user._id || user.id;
    setEditingId(userId);

    setFormData({
      name: user.name ?? "",
      email: user.email ?? "",
      password: "",
      role: user.role ?? "employee",
    });

    setError("");
    setSuccess("");

    window.scrollTo({ top: 0, behavior: "smooth" });
  };

  const handleDelete = async (id) => {
    const confirmed = window.confirm("Are you sure you want to delete this user?");

    if (!confirmed) return;

    setError("");
    setSuccess("");

    try {
      const response = await axios.delete(`${API}/api/user/${id}`, {
        withCredentials: true,
      });

      setSuccess(response.data.message || "User deleted successfully.");

      if (editingId === id) {
        resetForm();
      }

      revalidator.revalidate();
    } catch (error) {
      setError(error.response?.data?.message || "Failed to delete user.");
      console.error("Delete User Error:", error.response?.data || error.message);
    }
  };

  const handlePageChange = (page) => {
    if (page < 1 || page > totalPage || page === currentPage) return;

    setSearchParams((prev) => {
      const params = new URLSearchParams(prev);
      params.set("page", String(page));
      params.set("limit", String(limit));
      return params;
    });
  };

  const handleLimitChange = (e) => {
    const newLimit = Number(e.target.value);

    setSearchParams((prev) => {
      const params = new URLSearchParams(prev);
      params.set("page", "1");
      params.set("limit", String(newLimit));
      return params;
    });
  };

  const filteredUsers = useMemo(() => {
    return users.filter((user) => {
      const matchesSearch =
        user.name.toLowerCase().includes(search.toLowerCase()) ||
        user.email.toLowerCase().includes(search.toLowerCase());
      const matchesRole = roleFilter === "all" || user.role === roleFilter;
      return matchesSearch && matchesRole;
    });
  }, [users, search, roleFilter]);

  return (
    <main className="user-container">
      <section className="user-form-section">
        <div className="user-section-header">
          <h1>{editingId ? "Update User" : "Create User"}</h1>
          <p>
            {editingId
              ? "Edit the user details and save your changes."
              : "Add a user and assign the appropriate role."}
          </p>
        </div>

        <form className="user-form" onSubmit={handleSubmit}>
          <div className="user-form-group">
            <label htmlFor="name">Full Name</label>
            <input
              id="name"
              name="name"
              type="text"
              placeholder="Enter full name"
              value={formData.name}
              onChange={handleChange}
              autoComplete="name"
              required
            />
          </div>

          <div className="user-form-group">
            <label htmlFor="email">Email Address</label>
            <input
              id="email"
              name="email"
              type="email"
              placeholder="Enter email address"
              value={formData.email}
              onChange={handleChange}
              autoComplete="email"
              required
            />
          </div>

          <div className="user-form-group">
            <label htmlFor="password">
              {editingId ? "New Password" : "Password"}
            </label>
            <input
              id="password"
              name="password"
              type="password"
              placeholder={
                editingId
                  ? "Leave blank to keep current password"
                  : "Enter password"
              }
              value={formData.password}
              onChange={handleChange}
              autoComplete="new-password"
              minLength={8}
              required={!editingId}
            />
            <small>
              {editingId
                ? "Only fill this in if you want to change the password (min 8 characters)."
                : "Password must be at least 8 characters."}
            </small>
          </div>

          <div className="user-form-group">
            <label htmlFor="role">User Role</label>
            <select
              id="role"
              name="role"
              value={formData.role}
              onChange={handleChange}
              required
            >
              <option value="employee">Employee</option>
              <option value="manager">Manager</option>
              <option value="admin">Admin</option>
            </select>
          </div>

          {error && (
            <p className="user-message user-error" role="alert">{error}</p>
          )}

          {success && (
            <p className="user-message user-success" role="status">{success}</p>
          )}

          <div className="user-actions">
            <button
              type="submit"
              className="user-submit-btn"
              disabled={loading}
            >
              {loading
                ? "Saving..."
                : editingId
                  ? "Update User"
                  : "Create User"}
            </button>

            {editingId && (
              <button
                type="button"
                className="user-cancel-btn"
                onClick={resetForm}
                disabled={loading}
              >
                Cancel Edit
              </button>
            )}
          </div>
        </form>
      </section>

      <section className="user-list-section">
        <div className="user-section-header user-list-header">
          <div>
            <h2>User List</h2>
            <p>View and manage users and their roles.</p>
          </div>
        </div>

        <div className="user-toolbar">
          <input
            type="search"
            placeholder="Search by name or email..."
            aria-label="Search users"
            value={search}
            onChange={(e) => setSearch(e.target.value)}
          />

          <select
            aria-label="Filter users by role"
            value={roleFilter}
            onChange={(e) => setRoleFilter(e.target.value)}
          >
            <option value="all">All Roles</option>
            <option value="admin">Admin</option>
            <option value="manager">Manager</option>
            <option value="employee">Employee</option>
          </select>
        </div>

        <div className="user-table-wrapper">
          <table className="user-table">
            <thead>
              <tr>
                <th>User</th>
                <th>Role</th>
                <th>Created</th>
                <th>Action</th>
              </tr>
            </thead>

            <tbody>
              {
                filteredUsers.length > 0 ? (
                  filteredUsers.map((user) => {
                    const userId = user._id || user.id;
                    const isEditing = editingId === userId;

                    return (
                      <tr
                        key={userId}
                        className={isEditing ? "is-editing" : undefined}
                      >
                        <td>
                          <div className="user-profile">
                            <div className="user-avatar">
                              {user.name.charAt(0).toUpperCase()}
                            </div>

                            <div className="user-profile-details">
                              <strong>{user.name}</strong>
                              <span>{user.email}</span>
                            </div>
                          </div>
                        </td>

                        <td>
                          <span className={`user-role-badge role-${user.role}`}>
                            {user.role}
                          </span>
                        </td>

                        <td>
                          {user.createdAt
                            ? String(user.createdAt).slice(0, 10)
                            : "—"}
                        </td>

                        <td>
                          <details className="row-menu">
                            <summary className="row-menu-trigger" aria-label="Open menu">
                              <span></span><span></span><span></span>
                            </summary>

                            <div className="row-menu-panel" role="menu">
                              <button
                                type="button"
                                className="row-menu-item"
                                role="menuitem"
                                onClick={() => {/* view profile */ }}
                              >
                                <span className="row-menu-item-icon">👤</span>
                                <span>View profile</span>
                              </button>

                              <button
                                type="button"
                                className="row-menu-item"
                                role="menuitem"
                                onClick={() => handleEdit(user)}
                              >
                                <span className="row-menu-item-icon">✏️</span>
                                <span>Edit user</span>
                              </button>

                              <button
                                type="button"
                                className="row-menu-item"
                                role="menuitem"
                                onClick={() => {/* reset password */ }}
                              >
                                <span className="row-menu-item-icon">🔑</span>
                                <span>Reset password</span>
                              </button>

                              <button
                                type="button"
                                className="row-menu-item is-danger"
                                role="menuitem"
                                onClick={() => handleDelete(userId)}
                                aria-label={`Delete ${user.name}`}
                              >
                                <span className="row-menu-item-icon">🗑️</span>
                                <span>Delete</span>
                              </button>
                            </div>
                          </details>
                        </td>
                      </tr>
                    );
                  })
                ) : (
                  <tr>
                    <td colSpan="4" className="user-empty-state">
                      <div className="user-empty-icon">👥</div>
                      <strong>
                        {users.length === 0 ? "No users yet" : "No matching users"}
                      </strong>
                      <p>
                        {users.length === 0
                          ? "Create your first user using the form."
                          : "Try changing your search or role filter."}
                      </p>
                    </td>
                  </tr>
                )
              }
            </tbody>
          </table>
        </div>

        <div className="user-list-footer">
          <span>{totalUsers} total users</span>

          <div className="pagination-controls">
            <label htmlFor="users-limit">Rows per page:</label>

            <select
              id="users-limit"
              value={limit}
              onChange={handleLimitChange}
            >
              <option value={5}>5</option>
              <option value={10}>10</option>
              <option value={20}>20</option>
              <option value={50}>50</option>
            </select>

            <button
              type="button"
              disabled={currentPage <= 1}
              onClick={() => handlePageChange(currentPage - 1)}
            >
              Previous
            </button>

            <span>
              Page {currentPage} of {totalPage}
            </span>

            <button
              type="button"
              disabled={currentPage >= totalPage}
              onClick={() => handlePageChange(currentPage + 1)}
            >
              Next
            </button>
          </div>
        </div>
      </section>
    </main>
  );
};

export default User;
