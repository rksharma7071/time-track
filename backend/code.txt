import bcrypt from "bcrypt";
import jwt from "jsonwebtoken";
import User from "../models/user.model.js";
import { EMAIL_REGEX, normalizeEmail, validatePassword } from "../utils/validators.js";
import { blacklistToken, JWT_ERRORS } from "../utils/token.js";
import { toPublicUser } from "../utils/serializers.js";

const TOKEN_TTL_MS = 7 * 24 * 60 * 60 * 1000;

const cookieOptions = () => ({
  httpOnly: true,
  secure: process.env.NODE_ENV === "production",
  sameSite: "lax",
  path: "/",
});

const DUMMY_HASH = bcrypt.hashSync("timing-safe-dummy-password", 10);

export const register = async (req, res) => {
  const name = String(req.body.name ?? "").trim();
  const email = normalizeEmail(req.body.email);
  const { password } = req.body;

  if (!name || !email || !password) {
    return res.status(400).json({ message: "Name, email and password are required." });
  }

  if (!EMAIL_REGEX.test(email)) {
    return res.status(400).json({ message: "Please enter a valid email address." });
  }

  const passwordError = validatePassword(password);
  if (passwordError) {
    return res.status(400).json({ message: passwordError });
  }

  if (await User.exists({ email })) {
    return res.status(409).json({ message: "Email id already registered." });
  }

  const user = await User.create({
    name,
    email,
    password: await bcrypt.hash(password, 10),
    role: "employee",
  });

  return res.status(201).json({ message: "User created successfully.", user: toPublicUser(user) });
};

export const login = async (req, res) => {
  const email = normalizeEmail(req.body.email);
  const { password } = req.body;

  if (!email || typeof password !== "string" || !password) {
    return res.status(400).json({ message: "Email and password are required." });
  }

  const user = await User.findOne({ email });
  const passwordMatches = await bcrypt.compare(password, user?.password ?? DUMMY_HASH);

  if (!user || !passwordMatches) {
    return res.status(401).json({ message: "Invalid email or password." });
  }

  if (!user.isActive) {
    return res.status(403).json({ message: "Your account has been deactivated. Please contact an administrator." });
  }

  const token = jwt.sign({ _id: user._id, role: user.role }, process.env.JWT_SECRET, {
    expiresIn: "7d",
  });

  res.cookie("token", token, { ...cookieOptions(), maxAge: TOKEN_TTL_MS });

  return res.status(200).json({ message: "Login successful.", user: toPublicUser(user) });
};

export const logout = async (req, res) => {
  const token = req.cookies?.token;

  res.clearCookie("token", cookieOptions());

  if (token) {
    try {
      const decoded = jwt.verify(token, process.env.JWT_SECRET);
      await blacklistToken(token, decoded.exp);
    } catch (error) {
      if (!JWT_ERRORS.includes(error.name)) throw error;
    }
  }

  return res.status(200).json({ message: "Logout successful." });
};

export const changePassword = async (req, res) => {
  const { currentPassword, newPassword } = req.body;

  if (!currentPassword || !newPassword) {
    return res.status(400).json({ message: "Current password and new password are required." });
  }

  const passwordError = validatePassword(newPassword);
  if (passwordError) {
    return res.status(400).json({ message: passwordError });
  }

  if (currentPassword === newPassword) {
    return res.status(400).json({ message: "New password must be different from the current password." });
  }

  const user = await User.findById(req.user.id);

  if (!(await bcrypt.compare(currentPassword, user.password))) {
    return res.status(400).json({ message: "Current password is incorrect." });
  }

  user.password = await bcrypt.hash(newPassword, 10);
  user.passwordChangedAt = new Date();
  await user.save();

  await blacklistToken(req.token, req.tokenExp);
  res.clearCookie("token", cookieOptions());

  return res.status(200).json({ message: "Password changed successfully. Please login again." });
};

import Project, { PROJECT_STATUSES, TITLE_COLLATION } from "../models/project.model.js";
import Task from "../models/task.model.js";
import User from "../models/user.model.js";
import Permission from "../models/permission.model.js";
import { buildPagination, isId, parsePagination } from "../utils/validators.js";
import { accessibleProjectsFilter, canAccessProject, canManageProject } from "../utils/access.js";

const POPULATE = [

const parseDate = (value) => {
  const date = new Date(value);
  return Number.isNaN(date.getTime()) ? null : date;

const findTitleConflict = (title, excludeId) =>
  Project.findOne({ title, ...(excludeId && { _id: { $ne: excludeId } }) })
    .collation(TITLE_COLLATION)
    .select("title")
    .lean();

const validateManager = async (managerId) => {
  if (!isId(managerId)) return "Invalid manager ID.";

  const user = await User.findById(managerId).select("role isActive").lean();

  if (!user || !user.isActive) return "Manager does not exist or is inactive.";
  if (!["manager", "admin"].includes(user.role)) return "Selected user must be a manager or admin.";

  return null;
};

const validateMembers = async (members) => {
  if (!Array.isArray(members)) return { error: "Members must be an array of user IDs." };

  const ids = [...new Set(members.map(String))];

  if (ids.some((id) => !isId(id))) return { error: "One or more member IDs are invalid." };

  if (ids.length) {
    const found = await User.countDocuments({ _id: { $in: ids }, isActive: true });
    if (found !== ids.length) return { error: "One or more members do not exist or are inactive." };
  }

  return { ids };
};

export const createProject = async (req, res) => {
  const { title, description, type, status, clientName, manager, members = [], startDate, dueDate, totalHours } = req.body;
  const trimmedTitle = String(title ?? "").trim();

  if (!trimmedTitle || !manager || !startDate || totalHours === undefined || totalHours === null) {
    return res.status(400).json({ message: "Title, manager, start date and total hours are required." });
  }

  if (req.user.role === "manager" && String(manager) !== req.user.id) {
    return res.status(403).json({ message: "Managers can only create projects that they manage themselves." });
  }

  const managerError = await validateManager(manager);
  if (managerError) return res.status(400).json({ message: managerError });

  const { ids: memberIds, error: membersError } = await validateMembers(members);
  if (membersError) return res.status(400).json({ message: membersError });

  const start = parseDate(startDate);
  const due = dueDate ? parseDate(dueDate) : null;

  if (!start || (dueDate && !due)) {
    return res.status(400).json({ message: "Invalid project date." });
  }

  if (due && due < start) {
    return res.status(400).json({ message: "Due date cannot be before the start date." });
  }

  const hours = Number(totalHours);
  if (!Number.isFinite(hours) || hours <= 0) {
    return res.status(400).json({ message: "Total hours must be a positive number." });
  }

  const conflict = await findTitleConflict(trimmedTitle);
  if (conflict) {
    return res.status(409).json({ message: `Project "${conflict.title}" already exists.` });
  }

  const project = await Project.create({
    title: trimmedTitle,
    description,
    type,
    status,
    clientName,
    manager,
    members: memberIds,
    startDate: start,
    dueDate: due || undefined,
    totalHours: hours,
    remainingHours: hours,
    createdBy: req.user.id,
  });

  const createdProject = await Project.findById(project._id).populate(POPULATE).lean();

  return res.status(201).json({ message: "Project created successfully.", project: createdProject });
};

export const updateProject = async (req, res) => {
  const { id } = req.params;

  if (!isId(id)) {
    return res.status(400).json({ message: "Invalid project ID." });
  }

  const project = await Project.findById(id);

  if (!project) {
    return res.status(404).json({ message: "Project not found." });
  }

  if (!canManageProject(req.user, project)) {
    return res.status(403).json({ message: "You are not allowed to update this project." });
  }

  const { title, description, type, status, clientName, manager, members, startDate, dueDate, totalHours } = req.body;

  if (title !== undefined) {
    const trimmed = String(title).trim();
    if (!trimmed) return res.status(400).json({ message: "Title cannot be empty." });

    const conflict = await findTitleConflict(trimmed, id);
    if (conflict) {
      return res.status(409).json({ message: `Project "${conflict.title}" already exists.` });
    }
    project.title = trimmed;
  }

  if (description !== undefined) project.description = description;
  if (clientName !== undefined) project.clientName = clientName;
  if (type !== undefined) project.type = type;
  if (status !== undefined) project.status = status;

  if (manager !== undefined && String(manager) !== String(project.manager)) {
    if (req.user.role !== "admin") {
      return res.status(403).json({ message: "Only an admin can reassign the project manager." });
    }
    const managerError = await validateManager(manager);
    if (managerError) return res.status(400).json({ message: managerError });
    project.manager = manager;
  }

  if (members !== undefined) {
    const { ids, error } = await validateMembers(members);
    if (error) return res.status(400).json({ message: error });
    project.members = ids;
  }

  if (startDate !== undefined) {
    const start = parseDate(startDate);
    if (!start) return res.status(400).json({ message: "Invalid start date." });
    project.startDate = start;
  }

  if (dueDate !== undefined) {
    if (dueDate === null || dueDate === "") {
      project.dueDate = undefined;
    } else {
      const due = parseDate(dueDate);
      if (!due) return res.status(400).json({ message: "Invalid due date." });
      project.dueDate = due;
    }
  }

  if (project.dueDate && project.dueDate < project.startDate) {
    return res.status(400).json({ message: "Due date cannot be before the start date." });
  }

  if (totalHours !== undefined) {
    const parsed = Number(totalHours);

    if (!Number.isFinite(parsed) || parsed <= 0) {
      return res.status(400).json({ message: "Total hours must be a positive number." });
    }

    const used = await getProjectHoursUsed(project._id);

    if (parsed < used) {
      return res.status(400).json({ message: `Total hours cannot be less than the ${used} hour(s) already logged.` });
    }

    project.totalHours = parsed;
    project.remainingHours = parsed - used;
  }

  await project.save();

  const updatedProject = await Project.findById(id).populate(POPULATE).lean();

  return res.status(200).json({ message: "Project updated successfully.", project: updatedProject });
};

export const deleteProject = async (req, res) => {
  const { id } = req.params;

  if (!isId(id)) {
    return res.status(400).json({ message: "Invalid project ID." });
  }

  const project = await Project.findById(id).select("title");

  if (!project) {
    return res.status(404).json({ message: "Project not found." });
  }

  const taskCount = await Task.countDocuments({ project: id });

  if (taskCount > 0) {
    return res.status(409).json({
      message: `This project has ${taskCount} logged task(s). Set its status to "completed" or "cancelled" instead of deleting it.`,
    });
  }

  await Promise.all([
    project.deleteOne(),
    Permission.updateMany({ projects: id }, { $pull: { projects: id } }),
  ]);

  return res.status(200).json({ message: "Project deleted successfully." });
};

const pick = (source, keys) =>

const validateTaskEntry = async ({ user, projectId, employeeId, hours, date, excludeTaskId }) => {

  if (!day) return fail(400, "Invalid task date.");

  const latestAllowedDay = toUtcDay(Date.now());
  latestAllowedDay.setUTCDate(latestAllowedDay.getUTCDate() + 1);
  if (day > latestAllowedDay) return fail(400, "Hours cannot be logged for future dates.");

  const [project, employee] = await Promise.all([
    Project.findById(projectId).select("title status startDate totalHours manager members").lean(),
    User.findById(employeeId).select("role isActive").lean(),
  ]);

  if (!project) return fail(404, "Project does not exist.");
  if (!employee || !employee.isActive) return fail(404, "Employee does not exist or is inactive.");

  if (CLOSED_PROJECT_STATUSES.includes(project.status)) {
    return fail(400, `Hours cannot be logged on a ${project.status} project.`);
  }

  if (day < toUtcDay(project.startDate)) {
    return fail(400, "Task date cannot be before the project start date.");
  }

  const isSelf = String(employeeId) === user.id;
  if (!isSelf && !canManageProject(user, project)) {
    return fail(403, "You can only log hours for yourself on this project.");
  }

  const employeeOnProject = await canAccessProject({ id: String(employeeId), role: employee.role }, project);
  if (!employeeOnProject) {
    return fail(403, isSelf ? "You do not have access to this project." : "This employee is not assigned to the project.");
  }

  const [projectUsed, dayUsed] = await Promise.all([
    getProjectHoursUsed(project._id, excludeTaskId),
    getEmployeeDayHours(employeeId, day, excludeTaskId),
  ]);

  if (projectUsed + hours > project.totalHours) {
    const left = Math.max(0, project.totalHours - projectUsed);
    return fail(400, `Project's allocated hours would be exceeded. Only ${left} hour(s) remaining.`);
  }

  if (dayUsed + hours > MAX_HOURS_PER_DAY) {
    return fail(400, `Daily limit exceeded: ${dayUsed} hour(s) are already logged for this date.`);
  }

  return { project, day };
};

    conditions.push({ employee: toObjectId(employee) });

    if (!TASK_STATUSES.includes(status)) return res.status(400).json({ message: "Invalid status filter." });
    conditions.push({ status });

  if (type !== undefined) {
    if (!TASK_TYPES.includes(type)) return res.status(400).json({ message: "Invalid type filter." });
    conditions.push({ type });
  }

  if (from !== undefined || to !== undefined) {
    const range = {};

    if (from !== undefined) {
      const fromDay = toUtcDay(from);
      if (!fromDay) return res.status(400).json({ message: "Invalid 'from' date." });
      range.$gte = fromDay;
    }

    Task.find(filter).populate(POPULATE).sort({ date: -1, createdAt: -1 }).skip(skip).limit(limit),
    Task.countDocuments(filter),
    sumTaskHours(filter),
  ]);

  return res.status(200).json({
    message: "Tasks fetched successfully.",
    tasks,
    summary: { totalHours },
    pagination: buildPagination("totalTasks", totalTasks, page, limit),

import jwt from "jsonwebtoken";
import User from "../models/user.model.js";
import BlacklistedToken from "../models/blacklistedToken.model.js";
import { hashToken, JWT_ERRORS } from "../utils/token.js";

export const verifyToken = (allowedRoles = []) => {
  return async (req, res, next) => {
    const token = req.cookies?.token;

    if (!token) {
      return res.status(401).json({ message: "Authentication required." });
    }

    let decoded;
    try {
      decoded = jwt.verify(token, process.env.JWT_SECRET);
    } catch (error) {
      if (JWT_ERRORS.includes(error.name)) {
        return res.status(401).json({ message: "Invalid or expired token. Please login again." });
      }
      throw error;
    }

    const [isBlacklisted, user] = await Promise.all([
      BlacklistedToken.exists({ tokenHash: hashToken(token) }),
      User.findById(decoded._id).select("name email role isActive passwordChangedAt").lean(),
    ]);

    if (isBlacklisted) {
      return res.status(401).json({ message: "Session expired. Please login again." });
    }

    if (!user || !user.isActive) {
      return res.status(401).json({ message: "Your account is inactive or no longer exists." });
    }

    if (
      user.passwordChangedAt &&
      decoded.iat < Math.floor(new Date(user.passwordChangedAt).getTime() / 1000)
    ) {
      return res.status(401).json({ message: "Your password was changed. Please login again." });
    }

    if (allowedRoles.length && !allowedRoles.includes(user.role)) {
      return res.status(403).json({ message: "You do not have permission to access this resource." });
    }

    req.user = {
      id: String(user._id),
      name: user.name,
      email: user.email,
      role: user.role,
    };
    req.token = token;
    req.tokenExp = decoded.exp;

    return next();
  };
};

export const errorHandler = (err, req, res, next) => {
  if (err.type === "entity.parse.failed") {
    return res.status(400).json({ message: "Request body is not valid JSON." });
  }

  if (err.type === "entity.too.large") {
    return res.status(413).json({ message: "Request body is too large." });
  }

  if (err.name === "CastError") {
    return res.status(400).json({ message: `Invalid value for "${err.path}".` });
  }

  if (err.name === "ValidationError") {
    const message = Object.values(err.errors).map((e) => e.message).join(" ");
    return res.status(400).json({ message });
  }

  if (err.code === 11000) {
    const field = Object.keys(err.keyValue ?? {})[0] ?? "field";
    return res.status(409).json({ message: `A record with this ${field} already exists.` });
  }

  console.error(`[${req.method} ${req.originalUrl}]`, err);
  return res.status(500).json({ message: "Internal server error." });
};

import Permission from "../models/permission.model.js";
import Project from "../models/project.model.js";
import { toObjectId } from "./validators.js";

export const idOf = (value) => {
  const id = value?._id ?? value;
  return id == null ? null : String(id);
};

const sameId = (a, b) => {
  const x = idOf(a);
  const y = idOf(b);
  return Boolean(x && y && x === y);
};

export const accessibleProjectsFilter = async (user) => {
  if (user.role === "admin") return {};

  const userId = toObjectId(user.id);
  const permission = await Permission.findOne({ user: userId }).select("projects").lean();

  return {
    $or: [
      { manager: userId },
      { members: userId },
      { _id: { $in: permission?.projects ?? [] } },
    ],
  };
};

export const canAccessProject = async (user, project) => {
  if (!project) return false;
  if (user.role === "admin") return true;
  if (sameId(project.manager, user.id)) return true;
  if ((project.members ?? []).some((member) => sameId(member, user.id))) return true;

  const permitted = await Permission.exists({ user: user.id, projects: idOf(project) });
  return Boolean(permitted);
};

export const canManageProject = (user, project) =>
  Boolean(project) &&
  (user.role === "admin" || (user.role === "manager" && sameId(project.manager, user.id)));

export const visibleTasksFilter = async (user) => {
  if (user.role === "admin") return {};

  const userId = toObjectId(user.id);

  if (user.role === "manager") {
    const managedProjectIds = await Project.distinct("_id", { manager: userId });
    return { $or: [{ employee: userId }, { project: { $in: managedProjectIds } }] };
  }

  return { employee: userId };
};

export const canAccessTask = (user, task, project) => user.role === "admin" || sameId(task.employee, user.id) || canManageProject(user, project);

import Task from "../models/task.model.js";
import Project from "../models/project.model.js";
import { toObjectId } from "./validators.js";

export const sumTaskHours = async (match) => {
  const [result] = await Task.aggregate([
    { $match: match },
    { $group: { _id: null, total: { $sum: "$hours" } } },
  ]);
  return result?.total ?? 0;
};

const excludeTask = (taskId) => (taskId ? { _id: { $ne: toObjectId(taskId) } } : {});

export const getProjectHoursUsed = (projectId, excludeTaskId) =>
  sumTaskHours({ project: toObjectId(projectId), ...excludeTask(excludeTaskId) });

export const getEmployeeDayHours = (employeeId, day, excludeTaskId) => {
  const nextDay = new Date(day);
  nextDay.setUTCDate(nextDay.getUTCDate() + 1);

  return sumTaskHours({
    employee: toObjectId(employeeId),
    date: { $gte: day, $lt: nextDay },
    ...excludeTask(excludeTaskId),
  });
};

export const syncRemainingHours = async (projectId) => {
  const project = await Project.findById(projectId).select("totalHours").lean();
  if (!project) return;

  const used = await getProjectHoursUsed(projectId);

  await Project.updateOne(
    { _id: projectId },
    { $set: { remainingHours: Math.max(0, project.totalHours - used) } }
  );
};

import { createHash } from "crypto";
import BlacklistedToken from "../models/blacklistedToken.model.js";

export const JWT_ERRORS = ["JsonWebTokenError", "TokenExpiredError", "NotBeforeError"];

export const hashToken = (token) => createHash("sha256").update(token).digest("hex");

export const blacklistToken = async (token, exp) => {
  if (!token || !exp) return;
  const tokenHash = hashToken(token);

  await BlacklistedToken.updateOne(
    { tokenHash },
    { $setOnInsert: { tokenHash, expiresAt: new Date(exp * 1000) } },
    { upsert: true }
  );
};

import mongoose from "mongoose";

export const isId = (value) => value != null && mongoose.isObjectIdOrHexString(value);

export const toObjectId = (value) => new mongoose.Types.ObjectId(String(value));

export const EMAIL_REGEX = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

export const normalizeEmail = (email) => String(email ?? "").trim().toLowerCase();

export const validatePassword = (password) => {
  if (typeof password !== "string" || password.length < 8) {
    return "Password must be at least 8 characters long.";
  }
  if (password.length > 72) {
    return "Password must be at most 72 characters long.";
  }
  if (!/[A-Za-z]/.test(password) || !/\d/.test(password)) {
    return "Password must contain at least one letter and one number.";
  }
  return null;
};

export const isValidHours = (hours) =>
  Number.isFinite(hours) && hours >= 0.25 && hours <= 24 && Number.isInteger(hours * 4);

export const toUtcDay = (value) => {
  if (value === undefined || value === null || value === "") return null;
  const date = new Date(value);
  if (Number.isNaN(date.getTime())) return null;
  date.setUTCHours(0, 0, 0, 0);
  return date;
};

export const escapeRegex = (text) => text.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");

export const parsePagination = (query, { defaultLimit = 10, maxLimit = 100 } = {}) => {
  const page = Math.max(1, parseInt(query.page, 10) || 1);
  const limit = Math.min(maxLimit, Math.max(1, parseInt(query.limit, 10) || defaultLimit));
  return { page, limit, skip: (page - 1) * limit };
};

export const buildPagination = (totalKey, total, page, limit) => ({
  [totalKey]: total,
  currentPage: page,
  limit,
  totalPage: Math.ceil(total / limit),
  hasNextPage: page * limit < total,
  hasPrevPage: page > 1,
});
